Guard Contracts

Every guard implements a standard contract:

interface Guard {
  id:        string          // Unique guard identifier
  node_id:   string          // Node this guard is registered on
  trigger:   "STATE_ENTRY"   // When the guard fires
  evaluate:  (state, event) → GuardResult
}

type GuardResult =
  | { action: "PASS" }
  | { action: "BLOCK",    reason: string }
  | { action: "REDIRECT", target_node: string, event: Event }
