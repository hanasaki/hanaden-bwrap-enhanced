# Provision a new virtual root
bwrap-enhanced.sh provision \
  --host-real-root ~/my-sandbox

# Launch isolated shell (Tier 0 — max isolation)
bwrap-enhanced.sh start \
  --host-real-root ~/my-sandbox \
  -- /bin/bash

# Sandboxed Firefox (Tier 1 — GUI + Audio)
bwrap-enhanced.sh start \
  --host-real-root ~/my-sandbox \
  --wayland-passthrough \
  --audio-passthrough \
  -- firefox
